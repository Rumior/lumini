# Lumini web client

AI-powered business mentor and business-intelligence platform. This package is the **web client**.

> **Status: prototype (Stage 0 complete).** The interface is built, but all data is mock data held in
> `src/data/mock-data.ts` and browser `localStorage`. There is no backend, authentication, database,
> real AI, or real financial engine yet. See the build plan in the project docs for what comes next.

## Run

```bash
npm install
npm run dev        # http://localhost:3000
npm run build      # type-check + production build
npm run lint
```

Requires Node.js 20+.

## Structure

```
src/
  App.tsx                 Route table
  main.tsx                Entry point (BrowserRouter)
  config/navigation.ts    Route paths, nav items, titles
  context/                Workspace state shared by all views (prototype)
  components/app/         Layout, sidebar, top bar, modals
  components/ui/          shadcn/ui primitives (mostly unused so far)
  views/                  One file per screen
  data/mock-data.ts       ALL sample data - delete entries as real data replaces them
  types/                  Shared TypeScript types
  lib/                    Small utilities
```

## Routes

| Path | Screen |
|---|---|
| `/` | Overview |
| `/mentor` | Ask Lumini |
| `/business` | My business |
| `/finances` | Finances |
| `/plans` | Plans & actions |
| `/insights` | Insights |
| `/settings` | Settings |

## Known prototype limits

- The mentor chat returns keyword-matched canned replies.
- Financial maths runs in the browser using floating point (replaced in Stage 2 by a tested decimal engine).
- Health score, trends, forecast shape, notifications and reports are fabricated.
- Settings toggles do not change behaviour.

import { Navigate, Route, Routes } from 'react-router'
import { Suspense, lazy } from 'react'
import { WorkspaceProvider } from '@/context/WorkspaceProvider'
import { Layout } from '@/components/app/Layout'
import AuthView from '@/views/AuthView'
import { routePaths } from '@/config/navigation'
import { useAuth } from '@/context/AuthContext'
import { ErrorBoundary } from '@/components/ErrorBoundary'
import './App.css'

const HomeView = lazy(() => import('@/views/HomeView'))
const MentorView = lazy(() => import('@/views/MentorView'))
const BusinessView = lazy(() => import('@/views/BusinessView'))
const FinancesView = lazy(() => import('@/views/FinancesView'))
const PlansView = lazy(() => import('@/views/PlansView'))
const InsightsView = lazy(() => import('@/views/InsightsView'))
const SettingsView = lazy(() => import('@/views/SettingsView'))

function ProtectedRoute({ children }: { children: React.ReactNode }) {
  const { isAuthenticated } = useAuth()
  if (!isAuthenticated) {
    return <Navigate to="/auth" replace />
  }
  return <>{children}</>
}

function LoadingFallback() {
  return <div className="p-8 text-center">Loading...</div>
}

export default function App() {
  const { isAuthenticated } = useAuth()
  
  return (
    <ErrorBoundary>
      <Routes>
        <Route path="/auth" element={isAuthenticated ? <Navigate to={routePaths.home} replace /> : <AuthView />} />
        <Route element={
          <ProtectedRoute>
            <WorkspaceProvider>
              <Layout />
            </WorkspaceProvider>
          </ProtectedRoute>
        }>
          <Route index element={<Suspense fallback={<LoadingFallback />}><HomeView /></Suspense>} />
          <Route path={routePaths.mentor} element={<Suspense fallback={<LoadingFallback />}><MentorView /></Suspense>} />
          <Route path={routePaths.business} element={<Suspense fallback={<LoadingFallback />}><BusinessView /></Suspense>} />
          <Route path={routePaths.finances} element={<Suspense fallback={<LoadingFallback />}><FinancesView /></Suspense>} />
          <Route path={routePaths.plans} element={<Suspense fallback={<LoadingFallback />}><PlansView /></Suspense>} />
          <Route path={routePaths.insights} element={<Suspense fallback={<LoadingFallback />}><InsightsView /></Suspense>} />
          <Route path={routePaths.settings} element={<Suspense fallback={<LoadingFallback />}><SettingsView /></Suspense>} />
        </Route>
        <Route path="*" element={<Navigate to={routePaths.home} replace />} />
      </Routes>
    </ErrorBoundary>
  )
}

import { createContext, useContext, useState, useCallback, type ReactNode } from 'react'
import type { AuthState, User } from '@/types'

type AuthContextType = AuthState & {
  login: (email: string, password: string) => Promise<void>
  signup: (email: string, password: string, displayName: string) => Promise<void>
  logout: () => void
  updateUser: (updates: Partial<User>) => void
}

const AuthContext = createContext<AuthContextType | null>(null)

export function AuthProvider({ children }: { children: ReactNode }) {
  const [authState, setAuthState] = useState<AuthState>({
    user: null,
    isAuthenticated: false,
    isLoading: false,
  })

  const login = useCallback(async (email: string, _password: string) => {
    setAuthState({ user: null, isAuthenticated: false, isLoading: true })
    // Mock auth - replace with real API call
    await new Promise(resolve => setTimeout(resolve, 500))
    const user: User = {
      id: 'usr_' + Date.now(),
      email,
      displayName: email.split('@')[0],
      firstName: email.split('@')[0].split('.')[0],
      initials: email.substring(0, 2).toUpperCase(),
      role: 'owner',
    }
    setAuthState({ user, isAuthenticated: true, isLoading: false })
    localStorage.setItem('lumini_auth', JSON.stringify(user))
  }, [])

  const signup = useCallback(async (email: string, _password: string, displayName: string) => {
    setAuthState({ user: null, isAuthenticated: false, isLoading: true })
    await new Promise(resolve => setTimeout(resolve, 500))
    const user: User = {
      id: 'usr_' + Date.now(),
      email,
      displayName,
      firstName: displayName.split(' ')[0],
      initials: displayName.substring(0, 2).toUpperCase(),
      role: 'owner',
    }
    setAuthState({ user, isAuthenticated: true, isLoading: false })
    localStorage.setItem('lumini_auth', JSON.stringify(user))
  }, [])

  const logout = useCallback(() => {
    setAuthState({ user: null, isAuthenticated: false, isLoading: false })
    localStorage.removeItem('lumini_auth')
  }, [])

  const updateUser = useCallback((updates: Partial<User>) => {
    setAuthState(prev => {
      if (!prev.user) return prev
      const updated = { ...prev.user, ...updates }
      localStorage.setItem('lumini_auth', JSON.stringify(updated))
      return { ...prev, user: updated }
    })
  }, [])

  // Restore session from localStorage
  useState(() => {
    try {
      const stored = localStorage.getItem('lumini_auth')
      if (stored) {
        const user = JSON.parse(stored)
        setAuthState({ user, isAuthenticated: true, isLoading: false })
      }
    } catch {}
  })

  return (
    <AuthContext.Provider value={{ ...authState, login, signup, logout, updateUser }}>
      {children}
    </AuthContext.Provider>
  )
}

export function useAuth() {
  const ctx = useContext(AuthContext)
  if (!ctx) throw new Error('useAuth must be used inside <AuthProvider>')
  return ctx
}

import { createContext, useContext, useState, useCallback, useEffect, type ReactNode } from 'react'
import type { Organization, OrganizationMember } from '@/types'
import { useAuth } from './AuthContext'

type OrgContextType = {
  organization: Organization | null
  members: OrganizationMember[]
  isLoading: boolean
  createOrganization: (name: string) => Promise<void>
  inviteMember: (email: string, role: OrganizationMember['role']) => Promise<void>
  switchOrganization: (orgId: string) => void
}

const OrgContext = createContext<OrgContextType | null>(null)

export function OrganizationProvider({ children }: { children: ReactNode }) {
  const { user, isAuthenticated } = useAuth()
  const [organization, setOrganization] = useState<Organization | null>(null)
  const [members, setMembers] = useState<OrganizationMember[]>([])
  const [isLoading, setIsLoading] = useState(false)

  useEffect(() => {
    if (isAuthenticated && user && !organization) {
      // Create default org for user
      const defaultOrg: Organization = {
        id: 'org_' + Date.now(),
        name: user.displayName + "'s Business",
        plan: 'free',
        createdAt: new Date().toISOString(),
      }
      setOrganization(defaultOrg)
      setMembers([{
        organizationId: defaultOrg.id,
        userId: user.id,
        role: 'owner',
        status: 'active',
      }])
      localStorage.setItem('lumini_org', JSON.stringify(defaultOrg))
    }
  }, [isAuthenticated, user, organization])

  const createOrganization = useCallback(async (name: string) => {
    setIsLoading(true)
    await new Promise(r => setTimeout(r, 300))
    const org: Organization = {
      id: 'org_' + Date.now(),
      name,
      plan: 'free',
      createdAt: new Date().toISOString(),
    }
    setOrganization(org)
    if (user) {
      setMembers([{
        organizationId: org.id,
        userId: user.id,
        role: 'owner',
        status: 'active',
      }])
    }
    localStorage.setItem('lumini_org', JSON.stringify(org))
    setIsLoading(false)
  }, [user])

  const inviteMember = useCallback(async (email: string, role: OrganizationMember['role']) => {
    setIsLoading(true)
    await new Promise(r => setTimeout(r, 300))
    // Mock invitation
    console.log('Invited', email, 'as', role)
    setIsLoading(false)
  }, [])

  const switchOrganization = useCallback((orgId: string) => {
    console.log('Switch to org', orgId)
  }, [])

  return (
    <OrgContext.Provider value={{ organization, members, isLoading, createOrganization, inviteMember, switchOrganization }}>
      {children}
    </OrgContext.Provider>
  )
}

export function useOrganization() {
  const ctx = useContext(OrgContext)
  if (!ctx) throw new Error('useOrganization must be used inside <OrganizationProvider>')
  return ctx
}

      ...event,
      id: 'evt_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9),
      timestamp: new Date().toISOString(),
    }
    setEvents(prev => [newEvent, ...prev])
    console.log('[AUDIT]', newEvent.action, newEvent.resourceType)
  }, [])

    if (!filters) return events
    return events.filter(e => {
      return Object.entries(filters).every(([key, value]) => e[key as keyof AuditEvent] === value)
    })
  }, [events])

  return (
    <AuditContext.Provider value={{ events, logEvent, getEvents }}>
      {children}
    </AuditContext.Provider>
  )
}

export class ApiError extends Error {
  status: number
  constructor(message: string, status: number) {
    super(message)
    this.name = 'ApiError'
    this.status = status
  }
}

async function request<T>(path: string, options: RequestInit = {}): Promise<T> {
  const res = await fetch(`${API_BASE}${path}`, {
    headers: { 'Content-Type': 'application/json', ...(options.headers || {}) },
    credentials: 'include',
    ...options,
  })
  if (!res.ok) {
    throw new ApiError(res.statusText, res.status)
  }
  return res.json()
}

    this.answers[questionId] = answer
  }

    return { ...this.answers }
  }

    const keys = Object.keys(this.answers)

/**
 * Deterministic financial calculation engine for Lumini
 * Uses decimal-safe arithmetic and explicit formulas as per blueprint
 */

export function grossProfit(revenue: number, directCosts: number): number {
  return revenue - directCosts
}

export function grossMargin(revenue: number, directCosts: number): number {
  if (revenue === 0) return 0
  const gross = grossProfit(revenue, directCosts)
  return (gross / revenue) * 100
}

export function contributionPerUnit(sellingPrice: number, variableCost: number): number {
  return sellingPrice - variableCost
}

export function contributionMarginRatio(revenue: number, variableCosts: number): number {
  if (revenue === 0) return 0
  const contribution = revenue - variableCosts
  return contribution / revenue
}

export function breakEvenUnits(fixedCosts: number, contributionPerUnitVal: number): {
  units: number
  feasible: boolean
  warning?: string
} {
  if (contributionPerUnitVal <= 0) {
    return {
      units: 0,
      feasible: false,
      warning: 'Contribution per unit is zero or negative. Cannot break even with current pricing/costs.',
    }
  }
  if (fixedCosts <= 0) {
    return {
      units: 0,
      feasible: true,
    }
  }
  const units = Math.ceil(fixedCosts / contributionPerUnitVal)
  return {
    units,
    feasible: true,
  }
}

export function breakEvenRevenue(fixedCosts: number, contributionMarginRatioVal: number): {
  revenue: number
  feasible: boolean
  warning?: string
} {
  if (contributionMarginRatioVal <= 0) {
    return {
      revenue: 0,
      feasible: false,
      warning: 'Contribution margin ratio is zero or negative. Cannot break even with current structure.',
    }
  }
  if (fixedCosts <= 0) {
    return {
      revenue: 0,
      feasible: true,
    }
  }
  const revenue = fixedCosts / contributionMarginRatioVal
  return {
    revenue,
    feasible: true,
  }
}

export function operatingResult(revenue: number, directCosts: number, operatingExpenses: number): number {
  const gross = grossProfit(revenue, directCosts)
  return gross - operatingExpenses
}

export function netCashMovement(cashInflows: number, cashOutflows: number): number {
  return cashInflows - cashOutflows
}

export function cac(attributableSalesMarketingCost: number, newCustomersAcquired: number): {
  cac: number
  feasible: boolean
  warning?: string
} {
  if (newCustomersAcquired <= 0) {
    return {
      cac: 0,
      feasible: false,
      warning: 'No new customers acquired in this period.',
    }
  }
  return {
    cac: attributableSalesMarketingCost / newCustomersAcquired,
    feasible: true,
  }
}

import type { Action, Finance } from '@/types'
import type { HealthScore } from '@/types'

export function computeHealthScore(finance: Finance, actions: Action[], cashMonths: number): HealthScore {
  const gross = finance.revenue - finance.direct
  const marginPct = finance.revenue > 0 ? (gross / finance.revenue) * 100 : 0
  const marginScore = Math.min(100, Math.max(0, (marginPct / 40) * 100))
  const cashScore = Math.min(100, Math.max(0, (cashMonths / 6) * 100))
  const doneActions = actions.filter(a => a.done).length
  const actionScore = actions.length === 0 ? 0 : Math.min(100, (doneActions / actions.length) * 100)
  const pipelineScore = 70
  const overall = Math.round((marginScore * 0.3 + cashScore * 0.3 + actionScore * 0.25 + pipelineScore * 0.15))
  let level: HealthScore['level'] = 'ok'
  if (overall < 40) level = 'critical'
  else if (overall < 60) level = 'needs-attention'
  else if (overall >= 80) level = 'good'
  const recommendations: string[] = []
  if (marginPct < 30) recommendations.push('Improve gross margin - review pricing or direct costs')
  if (cashMonths < 3) recommendations.push('Low cash runway - focus on collections/cost reduction')
  if (actions.length > doneActions) recommendations.push('Complete planned actions to improve execution')
  if (recommendations.length === 0) recommendations.push('Business health looks stable')
  return {
    overall,
    breakdown: { cash: Math.round(cashScore), margin: Math.round(marginScore), actions: Math.round(actionScore), pipeline: pipelineScore },
    level,
    recommendations
  }
}

import type { TrendPoint } from '@/types'

export function generateTrends(months = 6): TrendPoint[] {
  const data: TrendPoint[] = []
  const baseRevenue = 280000
  for (let i = months - 1; i >= 0; i--) {
    const month = new Date(2026, 8 - i, 1).toLocaleString('en-US', { month: 'short' })
    const revenue = baseRevenue + i * 5000 + (Math.random() - 0.5) * 10000
    const expenses = revenue * 0.65
    const gross = revenue - expenses
    data.push({ month, revenue: Math.round(revenue), expenses: Math.round(expenses), gross: Math.round(gross) })
  }
  return data
}

import type { Action, Finance, Notification } from '@/types'

export function generateNotifications(finance: Finance, actions: Action[]): Notification[] {
  const notifications: Notification[] = []
  const gross = finance.revenue - finance.direct
  const marginPct = finance.revenue > 0 ? (gross / finance.revenue) * 100 : 0
  if (marginPct < 30) {
    notifications.push({
      id: 'n-margin',
      type: 'warning',
      title: 'Low gross margin',
      message: `Gross margin is ${marginPct.toFixed(1)}% - below target of 35%`,
      read: false,
      createdAt: new Date().toISOString()
    })
  }
  const overdue = actions.filter(a => !a.done && new Date(a.due) < new Date())
  if (overdue.length > 0) {
    notifications.push({
      id: 'n-overdue',
      type: 'info',
      title: 'Overdue actions',
      message: `${overdue.length} action(s) are past due`,
      read: false,
      createdAt: new Date().toISOString()
    })
  }
  if (notifications.length === 0) {
    notifications.push({
      id: 'n-ok',
      type: 'success',
      title: 'All good',
      message: 'No immediate issues detected',
      read: false,
      createdAt: new Date().toISOString()
    })
  }
  return notifications
}

import { useGo } from '@/hooks/use-go'
import { mockPromptChips, mockTopics } from '@/data/mock-data'

const topicIcons = { diagnose: TrendingUp, compare: SlidersHorizontal, numbers: Calculator }

export default function MentorView() {
  const go = useGo()
  const { messages, chatLoading, sendMessage, openNewAction } = useWorkspace()
  const [input, setInput] = useState('')
  const chatEnd = useRef<HTMLDivElement>(null)

  useEffect(() => { chatEnd.current?.scrollIntoView({ behavior: 'smooth' }) }, [messages, chatLoading])

  const send = (preset?: string) => {
    const text = (preset ?? input).trim()
    if (!text || chatLoading) return
    sendMessage(text)
    setInput('')
  }

  return (
    <section className="view mentor-view">
      <div className="page-heading compact">
        <div><p className="date-line">YOUR BUSINESS MENTOR</p><h1>Think it through, together.</h1><p>Advice grounded in your numbers, your context, and what is still unknown.</p></div>
        <EvidenceTag>Context connected</EvidenceTag>
      </div>
      <div className="mentor-layout">
        <div className="conversation">
          <div className="conversation-context"><Database size={15} /><span>Using September financials, business profile & active goals</span><button onClick={() => go('business')}>Inspect context</button></div>
          <div className="messages">
            {messages.map(m => (
              <div className={`message ${m.from}`} key={m.id}>
                {m.from === 'lumini' && <span className="mentor-seal"><Sparkles size={16} /></span>}
                <div>
                  <p>{m.text}</p>
                  {m.detail && <small><BookOpen size={13} />{m.detail}</small>}
                  {m.from === 'lumini' && m.id !== 1 && <button className="message-action" onClick={() => openNewAction(m.text.split('.')[0])}><Plus size={14} />Turn into an action</button>}
                </div>
              </div>
            ))}
            {chatLoading && (
              <div className="message lumini">
                <span className="mentor-seal"><Sparkles size={16} /></span>
                <div className="thinking"><i /><i /><i /><span>Reviewing your business context</span></div>
              </div>
            )}
            <div ref={chatEnd} />
          </div>
          <div className="prompt-chips">{mockPromptChips.map(p => <button key={p} onClick={() => send(p)}>{p}</button>)}</div>
          <form className="chat-form" onSubmit={e => { e.preventDefault(); send() }}>
            <textarea value={input} onChange={e => setInput(e.target.value)} onKeyDown={e => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); send() } }} placeholder="Ask about a decision, challenge, or opportunityâ€¦" aria-label="Message Lumini" rows={2} />
            <div>
              <span>Enter to send Â· Shift + Enter for a new line</span>
              <button className="send-btn" type="submit" disabled={!input.trim() || chatLoading} aria-label="Send message"><Send size={18} /></button>
            </div>
          </form>
          <p className="ai-note">Lumini can make mistakes. Review assumptions before consequential decisions.</p>
        </div>
        <aside className="mentor-aside">
          <p className="eyebrow">A BETTER CONVERSATION</p>
          <h3>Start with what matters.</h3>
          {mockTopics.map(x => {
            const Icon = topicIcons[x.key]
            return (
              <button className="topic-link" key={x.key} onClick={() => send(x.q)}>
                <Icon size={19} /><span><strong>{x.title}</strong><small>{x.desc}</small></span><ArrowRight size={15} />
              </button>
            )
          })}
          <div className="mentor-principle"><ShieldCheck size={20} /><p><strong>Context before advice.</strong><br />Lumini asks when a missing fact could change the answer.</p></div>
        </aside>
      </div>
    </section>
  )
}
